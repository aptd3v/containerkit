containerkit

The Docker stack as a program, in Go or TSX

nginx-example is serving this page on port 8080

Go

examples/nginx writes the Dockerfile, builds nginx-example:latest, and brings the service up. Setters are dot-imported from pkg/kit.

df := nginxDockerfile()
cli, err := Connect(FromEnv())
defer cli.Close()

ctx := context.Background()
err = BuildImage(ctx, cli.Engine(),
    df,
    Source("./examples/nginx"),
    Image("nginx-example:latest"),
)

app := NewCompose("nginx-example").WithService(NewService(
    Name("nginx"),
    Image("nginx-example:latest"),
    Command("nginx", "-g", "daemon off;"),
    Publish(Protocol(TCP), HostIP("0.0.0.0"), HostPort(8080), ContainerPort(80)),
    Health(
        Test("CMD-SHELL", "curl -f http://localhost:80 || exit 1"),
        Interval(10*time.Second),
        Timeout(5*time.Second),
        Retries(3),
    ),
    Memory(100 << 20),
    Develop(WatchFiles(
        Action(Sync),
        Path("./src"),
        Target("/app"),
        Ignore("tmp"),
    )),
    Build(Source("./examples/nginx"), df),
))
err = app.Up(ctx, cli.Composer(), Detach(true), RemoveOrphans(true))

func nginxDockerfile() *Dockerfile {
    return NewDockerfile().
        From("nginx", "latest").
        Copy("nginx.conf", "/etc/nginx/nginx.conf").
        Copy("index.html", "/usr/share/nginx/html/index.html").
        Expose("80").
        Command("nginx", "-g", "daemon off;")
}

TSX

tsx/examples/nginx builds the same files. The session stays up, waits until the container is healthy, and takes the project down on Ctrl+C.

const dockerfile = (
    <Dockerfile>
        <From image="nginx" tag="alpine" />
        <Copy src="nginx.conf" dest="/etc/nginx/nginx.conf" />
        <Copy src="index.html" dest="/usr/share/nginx/html/index.html" />
        <Expose port="80" />
        <Command executable="nginx" args={["-g", "daemon off;"]} />
    </Dockerfile>
);

function App() {
    useShutdown({ removeOrphans: true });
    return (
        <Compose name="nginx-example">
            <Service name="nginx" image="nginx-example:latest">
                <Build source={context}>{dockerfile}</Build>
                <Publish
                    protocol={proto.TCP}
                    hostIP="127.0.0.1"
                    hostPort={8080}
                    containerPort={80}
                />
                <Health
                    test={["CMD-SHELL", "wget -q -O /dev/null http://127.0.0.1/ || exit 1"]}
                    interval="5s"
                    timeout="3s"
                    retries={5}
                />
            </Service>
        </Compose>
    );
}

export default root(
    {
        fromEnv: true,
        onExit: "down",
        wait: true,
        onUp: () => console.log("nginx at http://127.0.0.1:8080"),
    },
    App,
);
